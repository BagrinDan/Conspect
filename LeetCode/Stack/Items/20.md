
    > Суть: Понять если строка состоящаяя из скобок валидна

    > Как решить: Stack & HashMap

    > Код:
        Java
            class Solution {
                public boolean isValid(String s) {
                    Map<Character, Character> map = Map.of(
                        '}' , '{',
                        ')' , '(',
                        ']' , '['
                    );

                    Stack<Character> stack = new Stack<>(); 

                    for(int i = 0; i < s.length(); i++){ 
                        char curr = s.charAt(i);
                        
                        if(map.containsKey(curr)){ 
                            if(stack.isEmpty()){
                                return false;
                            }

                            char top = stack.pop(); 

                            if(top != map.get(curr)){ 
                                return false;
                            }
                        } else{
                            stack.push(curr);
                        }
                    } 
                    return stack.isEmpty();
                }
            }

=========================================
ШПАРГАЛКА ПО НАСТРОЙКЕ МАРШРУТИЗАТОРА (ROUTER)
=========================================

[ Базовая настройка и безопасность ]
    configure terminal           - Вход в глобальный режим конфигурации
        hostname R1                  - Назначение имени маршрутизатора
        enable secret class          - Защищенный пароль для перехода в привилегированный режим (enable)
        service password-encryption  - Шифрование всех открытых паролей в конфиге (Type 7)
        security passwords min-length 14 - Установка минимальной длины пароля
        login block-for 120 attempts 3 within 60 - Защита от брутфорса: блок на 120 сек после 3 ошибок за 60 сек
        banner motd # TEXT #         - Установка баннера-предупреждения при входе

[ Настройка линий управления (Console и VTY) ]
    line con 0                   - Настройка физического консольного порта
        password cisco             - Установка пароля
        login                      - Включение проверки пароля при входе
        exec-timeout 4 0           - Автоотключение сессии через 4 мин 0 сек бездействия
    exit

    line vty 0 4                 - Настройка виртуальных линий (для сетевого доступа SSH/Telnet)
        password cisco             - Установка пароля (если не используется локальная база)
        login local                - Включение авторизации по локальной базе пользователей
        transport input ssh        - Разрешение подключений только по протоколу SSH
        exec-timeout 4 0           - Автоотключение сессии через 4 мин
    exit

[ Глобальные сетевые параметры ]
    ipv6 unicast-routing         - Включение маршрутизации IPv6-трафика (без нее IPv6 не маршрутизируется)

[ Настройка интерфейсов (LAN, WAN, Loopback) ]
    interface GigabitEthernet 0/0/0 - Настройка стандартного LAN-интерфейса
        description Link to LAN 1  - Текстовое описание интерфейса
        ip address 192.168.10.1 255.255.255.0 - Назначение IPv4-адреса и маски
        ipv6 address 2001:db8:acad:1::1/64    - Назначение глобального IPv6-адреса
        ipv6 address fe80::1 link-local       - Назначение локального IPv6-адреса (Link-Local)
        no shutdown                - Физическое включение интерфейса
    exit 

    interface Serial 0/0/0       - Настройка последовательного интерфейса (устаревший WAN-линк)
        description Link to R2
        ip address 209.165.200.225 255.255.255.252
        ipv6 address 2001:db8:acad:3::225/64
        no shutdown
    exit

    interface Loopback 0         - Создание виртуального интерфейса (всегда активен, используется для тестов/маршрутизации)
        ip address 10.0.0.1 255.255.255.0
    exit

[ Управление временем и сохранение конфигурации ]
    clock set 16:02:55 7 Sep 2026 - Установка времени и даты (команда выполняется в привилегированном режиме #)
    copy running-config startup-config - Сохранение настроек в энергонезависимую память (NVRAM)
